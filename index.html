require("dotenv").config(); const express = require("express"); const mongoose = require("mongoose"); const cors = require("cors"); const app = express(); app.use(cors()); app.use(express.json()); const PORT = process.env.PORT || 3000; mongoose .connect(process.env.MONGO_URI) .then(() => { console.log("✅ MongoDB connected successfully"); }) .catch((error) => { console.log("❌ MongoDB Error:", error); }); const feedbackSchema = new mongoose.Schema( { agentName: String, agentEmail: String, rating: Number, comment: String, country: { type: String, default: "Unknown" }, submittedAtIST: { type: String, default: "" } }, { timestamps: true } ); const Feedback = mongoose.model("Feedback", feedbackSchema); app.get("/", (req, res) => { res.send("CSAT backend is running"); }); app.post("/submit", async (req, res) => { try { const { agentName, agentEmail, rating, comment, country, submittedAtIST } = req.body; if (!agentEmail || !rating) { return res.status(400).json({ success: false, message: "Agent email and rating are required" }); } const istTime = submittedAtIST || new Date().toLocaleString("en-IN", { timeZone: "Asia/Kolkata", day: "2-digit", month: "2-digit", year: "numeric", hour: "2-digit", minute: "2-digit", second: "2-digit", hour12: true }); const feedback = new Feedback({ agentName, agentEmail, rating, comment: comment || "", country: country || "Unknown", submittedAtIST: istTime }); await feedback.save(); res.json({ success: true, message: "Feedback submitted successfully", savedData: { agentName, agentEmail, rating, country: country || "Unknown", submittedAtIST: istTime } }); } catch (error) { console.log("Submit Error:", error); res.status(500).json({ success: false, message: "Server Error" }); } }); app.listen(PORT, () => { console.log(`Server running on port ${PORT}`); });
